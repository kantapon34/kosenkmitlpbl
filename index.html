// EMG Signal Classifier - KOSEN-KMITL Bionic Hand (.c / .py / .html)
#include 
Servo thumb, indexPin, middlePin;

void setup() {
  Serial.begin(115200);
  thumb.attach(18);
  indexPin.attach(19);
}

void loop() {
  int emgVal = analogRead(34);
  if (emgVal > 1500) {
    thumb.write(90); // Grip gesture
  } else {
    thumb.write(0);  // Relaxed gesture
  }
  delay(20);
}
